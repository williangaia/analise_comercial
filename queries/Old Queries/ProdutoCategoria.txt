--ProdutoCategoria
SELECT
    P.SEQPRODUTO,
    P.SEQFAMILIA,
    F.FAMILIA,
    P.DESCCOMPLETA AS "PRODUTO",
    TO_CHAR(P.DTAHORINCLUSAO, 'YYYY-MM-DD') AS "DTAINCLUSAOPROD",
    TO_CHAR(P.DTAHORALTERACAO, 'YYYY-MM-DD') AS "DTAALTERACAOPROD",
    P.USUARIOALTERACAO AS "USUALTERACAOPROD",
    TO_CHAR(DC.DATAHORAALTERACAO, 'YYYY-MM-DD') AS "DTAALTERACAOFAMDIVCATEG",
    DC.USUARIOALTERACAO AS "USUALTERACAOFAMDIVCATEG",
    DC.STATUS,
    C.SEQCATEGORIA,
    C.CAMINHOCOMPLETO AS "CATEGORIA",
    TO_CHAR(C.DTAHORALTERACAO, 'YYYY-MM-DD') AS "DTAALTERACAOCATEG",
    TO_CHAR(A.DTAHORINCLUSAO, 'YYYY-MM-DD') AS "DTAINCLUSAOFAMDIV",
    TO_CHAR(A.DTAHORALTERACAO, 'YYYY-MM-DD') AS "DTAALTERACAOFAMDIV",
    A.USUARIOALTERACAO AS "USUALTERACAOFAMDIV",
    A.PADRAOEMBTRANSF,
    A.PADRAOEMBCOMPRA,
    CP.COMPRADOR,
    M.MARCA,
    FP.NOMERAZAO,
    dc.nrodivisao,
    fp.seqpessoa as "seqfornecedor"
FROM 
    consinco.map_produto P
JOIN 
    consinco.map_familia F ON P.SEQFAMILIA = F.SEQFAMILIA
JOIN 
    consinco.map_famdivisao A ON F.SEQFAMILIA = A.SEQFAMILIA
JOIN 
    consinco.map_famdivcateg DC ON A.SEQFAMILIA = DC.SEQFAMILIA AND DC.NRODIVISAO = A.NRODIVISAO
JOIN 
    consinco.map_categoria C ON DC.SEQCATEGORIA = C.SEQCATEGORIA AND DC.NRODIVISAO = C.NRODIVISAO AND DC.STATUS = C.STATUSCATEGOR
JOIN 
    consinco.max_comprador CP ON A.SEQCOMPRADOR = CP.SEQCOMPRADOR
JOIN
    consinco.map_marca M ON F.SEQMARCA = M.SEQMARCA
JOIN 
    consinco.MAP_FAMFORNEC FF ON F.SEQFAMILIA = FF.SEQFAMILIA
JOIN 
    consinco.GE_PESSOA FP ON FF.Seqfornecedor = FP.Seqpessoa
WHERE 
    DC.NRODIVISAO in (1, 2)
    AND C.NIVELHIERARQUIA = 4
    AND C.STATUSCATEGOR = 'A'
    AND FF.PRINCIPAL = 'S' 
    AND P.DTAHORALTERACAO = TRUNC(SYSDATE) - 1
